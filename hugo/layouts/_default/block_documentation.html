{{ define "main" }} {{- /* A rule-editor block. The page itself holds almost nothing: its description is written by the generator into docs/official.md, and whatever the community learned goes into docs/extra.md beside it. The notes section renders even when that file is untouched — a slot nobody has filled yet is still the page saying it can grow. */ -}} {{ $html := .Content }} {{ $indexFile := .File.Path }} {{ $dir := strings.TrimSuffix .File.LogicalName $indexFile }} {{ $extraFile := path.Join $dir "docs/extra.md" }} {{ $officialFile := path.Join $dir "docs/official.md" }} {{ if os.FileExists $officialFile }} {{ $html = printf "%s%s" $html (os.ReadFile $officialFile | markdownify) }} {{ end }} {{- /* extra.md ships holding one comment and nothing else; that comment is not a note, so it must not silence the empty state. */ -}} {{- $raw := "" -}} {{- if os.FileExists $extraFile }}{{ $raw = trim (os.ReadFile $extraFile) " \n\t" }}{{ end -}} {{- $placeholder := trim (printf "" .Page.Title) " \n\t" -}} {{- $hasNotes := and $raw (ne $raw $placeholder) -}} {{- $notes := slice ("## Community notes" | markdownify) -}} {{- if $hasNotes -}} {{- $notes = $notes | append ($raw | $.Page.RenderString) -}} {{- else -}} {{- $notes = $notes | append (partial "block-notes-empty" $.Page) -}} {{- end -}} {{ $html = printf "%s%s" $html (delimit $notes "\n\n") }} {{ partial "doc-shell" (dict "page" . "article" $html) }} {{ end }}
